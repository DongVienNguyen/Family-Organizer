import {test,expect} from '/workspace/Family-Organizer/node_modules/@playwright/test/index.mjs';
import {installOfflineNetworkBoundary} from '/workspace/Family-Organizer/tests/e2e/offline-network-boundary.ts';
test('W14 actual viewport evidence shows budget actual and forecast in both themes',async({page,context,baseURL})=>{
 const external:string[]=[];await installOfflineNetworkBoundary(context,baseURL!,external);await context.addInitScript(()=>localStorage.setItem('family_lang','vi'));await page.setViewportSize({width:390,height:844});const login=await page.request.post('/api/auth/login',{data:{username:'wallet-browser-member',password:'synthetic-browser-password'}});expect(login.status()).toBe(200);const token=(await login.json()).token;await context.setStorageState({cookies:[],origins:[{origin:baseURL!,localStorage:[{name:'family_token',value:token},{name:'family_lang',value:'vi'}]}]});await page.goto('/?tab=finance');const today=await page.evaluate(()=>new Intl.DateTimeFormat('en-CA',{timeZone:'Asia/Ho_Chi_Minh',year:'numeric',month:'2-digit',day:'2-digit'}).format(new Date()));
 const post=async(path:string,body:unknown)=>{const r=await page.request.post('/api/wallet/'+path,{headers:{Authorization:'Bearer '+token,'x-request-id':crypto.randomUUID()},data:body});expect(r.status()).toBe(200);return r.json();};const account=(await post('accounts',{expectedRevision:0,input:{name:'Ví gia đình mẫu',type:'cash',currency:'VND',openingMinor:'1000000',openingDate:today,accessMode:'private'}})).account;
 await post('records',{expectedRevision:0,input:{kind:'expense',accountId:account.id,money:{minor:'600000',currency:'VND'},effectiveDate:today,timezone:'Asia/Ho_Chi_Minh',note:'Chi tiêu mẫu'}});await post('schedules',{expectedRevision:0,draft:{name:'Khoản chi dự kiến mẫu',record:{kind:'expense',accountId:account.id,money:{minor:'250000',currency:'VND'},effectiveDate:today,timezone:'Asia/Ho_Chi_Minh',note:'Kế hoạch mẫu',occurredAt:null,reviewedAt:null,payee:'',paymentMethod:'',settlementStatus:'cleared',categoryId:null,labelIds:[]},recurrence:{anchorDate:today,timezone:'Asia/Ho_Chi_Minh',frequency:'monthly',interval:1,monthEndPolicy:'clamp'},mode:'manual',paused:false,reminderDays:[]}});await post('budgets',{expectedRevision:0,draft:{name:'Ngân sách gia đình mẫu',limit:{minor:'1000000',currency:'VND'},timezone:'Asia/Ho_Chi_Minh',filter:{accountIds:[account.id]},carryPolicy:'positive',alertThresholds:[]}});
 await page.getByTestId('wallet-home').getByRole('button',{name:'Lập kế hoạch',exact:true}).click();const p=page.getByTestId('wallet-budgets');await p.getByRole('button',{name:'Ngân sách gia đình mẫu',exact:true}).click();await expect(p.getByTestId('budget-remaining')).toContainText('400.000');await expect(p.getByTestId('budget-expectedBalance')).toContainText('150.000');for(const theme of ['light','dark']){await page.evaluate(t=>document.documentElement.classList.toggle('dark',t==='dark'),theme);const heading=p.getByRole('heading',{name:'Ngân sách và dự báo',exact:true});await heading.scrollIntoViewIfNeeded();await expect(heading).toBeVisible();await page.screenshot({path:`/workspace/family-evidence/w14-visual-artifacts/w14-budget-viewport-${theme}-header.png`});await p.getByTestId('budget-remaining').scrollIntoViewIfNeeded();await expect(p.getByTestId('budget-remaining')).toBeVisible();await page.screenshot({path:`/workspace/family-evidence/w14-visual-artifacts/w14-budget-viewport-${theme}-report.png`});expect(await p.evaluate(n=>n.scrollWidth<=n.clientWidth)).toBe(true);}expect(external).toEqual([]);
});
